cmake_minimum_required(VERSION 3.10)

# CXX is needed for Dear ImGui / cimgui, which are C++; everything else in
# this project stays C.
project(PBR_Renderer VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(OpenGL REQUIRED)

add_subdirectory(deps/glfw)
add_subdirectory(deps/glad)
add_subdirectory(deps/cglm)
include(cmake/cimgui.cmake)

add_executable(PBR_Renderer
    src/main.c
    src/App.c
    src/Core/FileIO.c
    src/Core/Paths.c
    src/Core/Image.c
    src/Renderer/Renderer.c
    src/RHI/RHI.c
    src/RHI/OpenGL/RHI_GL.c
    src/RHI/OpenGL/RHI_GL_Buffer.c
    src/RHI/OpenGL/RHI_GL_Shader.c
    src/RHI/OpenGL/RHI_GL_Texture.c
    src/RHI/OpenGL/RHI_GL_Framebuffer.c
    src/Platform/Window.c
    src/Scene/Camera.c
    src/UI/ImGuiLayer.c
)

target_link_libraries(PBR_Renderer PRIVATE
    glfw
    glad
    OpenGL::GL
    cglm
    cimgui
)

# stb_image is a single vendored header (no CMakeLists of its own, unlike
# the git-submodule deps above), so it just needs to be on the include path.
target_include_directories(PBR_Renderer PRIVATE deps/stb)

# Absolute path to assets/ baked into the binary, so asset_path() resolves
# correctly regardless of the process's current working directory.
target_compile_definitions(PBR_Renderer PRIVATE
    ASSETS_ROOT="${CMAKE_SOURCE_DIR}/assets"
)
